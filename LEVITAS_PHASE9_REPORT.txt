import os
import math
import struct
import collections

def tokenize(text):
    tokens = []
    in_tag = False
    for char in text:
        if char == '<':
            in_tag = True
            continue
        if char == '>':
            in_tag = False
            continue
        if not in_tag and ord(char) > 31 and ord(char) < 127:
            tokens.append((ord(char) % 127) + 1)
    return tokens

def analyze_manifold(filepath):
    with open(filepath, 'r') as f:
        doc_lines = f.readlines()
    
    all_tokens = []
    line_token_sets = []
    for line in doc_lines:
        line_tokens = tokenize(line)
        if line_tokens:
            all_tokens.extend(line_tokens)
            line_token_sets.append(line_tokens)

    # 1. Conditional Entropy (Markov Chain)
    transitions = collections.defaultdict(lambda: collections.defaultdict(int))
    for i in range(len(all_tokens) - 1):
        transitions[all_tokens[i]][all_tokens[i+1]] += 1
    
    high_lock_sequences = []
    for t1, next_tokens in transitions.items():
        total = sum(next_tokens.values())
        for t2, count in next_tokens.items():
            prob = count / total
            if prob > 0.9:
                high_lock_sequences.append((t1, t2, prob))

    # 2. Zipfian Distribution
    freq = collections.Counter(all_tokens)
    rank_table = sorted(freq.items(), key=lambda x: x[1], reverse=True)

    # 3. Word-Length (Tokens per whitespace)
    # Whitespace in neutral mapping is 33 ((ord(' ') % 127) + 1)
    words = []
    current_word = []
    for t in all_tokens:
        if t == 33:
            if current_word:
                words.append(len(current_word))
                current_word = []
        else:
            current_word.append(t)
    if current_word:
        words.append(len(current_word))
    word_length_dist = collections.Counter(words)

    # 4. Positional Variance
    initial_counts = collections.Counter()
    final_counts = collections.Counter()
    for line_tokens in line_token_sets:
        if line_tokens:
            initial_counts[line_tokens[0]] += 1
            final_counts[line_tokens[-1]] += 1
    
    total_lines = len(line_token_sets)
    pos_variance = {}
    for t in set(all_tokens):
        p_initial = initial_counts[t] / total_lines
        p_final = final_counts[t] / total_lines
        pos_variance[t] = (p_initial, p_final)

    # Header Tokens (Initial probability > 0.8)
    header_tokens = [t for t, (pi, pf) in pos_variance.items() if pi > 0.8]

    # 5. Generate Signature Binary
    # Format: [TransitionCount][(T1, T2, Count)][RankCount][(T, Freq)][WordLenCount][(Len, Freq)][PosCount][(T, PI, PF)]
    signature_path = "/home/phil/.gemini/antigravity/scratch/levitas-engine/VOYNICH_SIGNATURE.bin"
    with open(signature_path, 'wb') as bin_file:
        # Transitions
        bin_file.write(struct.pack('I', len(transitions)))
        for t1, nexts in transitions.items():
            bin_file.write(struct.pack('I', t1))
            bin_file.write(struct.pack('I', len(nexts)))
            for t2, count in nexts.items():
                bin_file.write(struct.pack('II', t2, count))
        
        # Rankings
        bin_file.write(struct.pack('I', len(rank_table)))
        for t, f in rank_table:
            bin_file.write(struct.pack('II', t, f))

        # Word Lengths
        bin_file.write(struct.pack('I', len(word_length_dist)))
        for l, f in word_length_dist.items():
            bin_file.write(struct.pack('II', l, f))

        # Positional Variance
        bin_file.write(struct.pack('I', len(pos_variance)))
        for t, (pi, pf) in pos_variance.items():
            bin_file.write(struct.pack('Iff', t, pi, pf))

    print(f"Signature generated: {signature_path}")
    print("\n[HIGH-LOCK SEQUENCES]")

    print("\n[ZIPFIAN RANK-FREQUENCY (Top 5)]")
    for t, f in rank_table[:5]:
        print(f"ID {t}: {f}")

    print("\n[POSITIONAL HEADERS]")
    for h in header_tokens:
        print(f"ID {h} - Initial Pred: {pos_variance[h][0]:.2%}")
